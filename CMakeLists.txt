	target_link_options(stories-options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined)

	target_link_options(stories-options INTERFACE -fsanitize=thread)

# Hardening applies to the shipped server and its tests, not to sanitizer or fuzz builds,
# which need undisturbed instrumentation. Toolchains differ (Apple's linker has no -z
# options and -fcf-protection is x86-only), so each option is probed and skipped if unsupported.
add_library(stories-hardening INTERFACE)
set(stories_pie OFF)
if(STORIES_HARDENING AND NOT STORIES_SANITIZERS AND NOT STORIES_THREAD_SANITIZER)
	foreach(stories_flag IN ITEMS -fstack-protector-strong -fstack-clash-protection -fcf-protection=full)
		string(MAKE_C_IDENTIFIER "STORIES_HAVE${stories_flag}" stories_probe)
		cmake_push_check_state(RESET)
		set(CMAKE_REQUIRED_FLAGS "-Werror ${stories_flag}")
		check_c_source_compiles("int main(void) { return 0; }" ${stories_probe})
		cmake_pop_check_state()
		if(${stories_probe})
			target_compile_options(stories-hardening INTERFACE ${stories_flag})
		endif()
	endforeach()
	foreach(stories_flag IN ITEMS -Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack)
		string(MAKE_C_IDENTIFIER "STORIES_HAVE${stories_flag}" stories_probe)
		cmake_push_check_state(RESET)
		set(CMAKE_REQUIRED_LINK_OPTIONS ${stories_flag})
		check_c_source_compiles("int main(void) { return 0; }" ${stories_probe})
		cmake_pop_check_state()
		if(${stories_probe})
			target_link_options(stories-hardening INTERFACE ${stories_flag})
		endif()
	endforeach()
	# Fortification needs optimization; glibc warns at -O0. Unset first so a distribution default of another level cannot conflict.
	target_compile_options(stories-hardening INTERFACE "$<$<NOT:$<CONFIG:Debug>>:-U_FORTIFY_SOURCE>" "$<$<NOT:$<CONFIG:Debug>>:-D_FORTIFY_SOURCE=2>")
	include(CheckPIESupported)
	check_pie_supported(LANGUAGES C CXX)
	# stories-core supplies C++ sources, so the server links with the C++ driver.
	if(CMAKE_CXX_LINK_PIE_SUPPORTED)
		set(stories_pie ON)
	endif()
endif()

if(STORIES_FUZZING)
	enable_testing()
	set(stories_fuzz_flags -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined -UNDEBUG -g -O1)
	set(stories_fuzz_link_flags -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=undefined)
	cmake_push_check_state(RESET)
	set(CMAKE_REQUIRED_FLAGS "-fsanitize=fuzzer-no-link,address,undefined")
	set(CMAKE_REQUIRED_LINK_OPTIONS ${stories_fuzz_link_flags})
	check_c_source_compiles("#include <stdint.h>\n#include <stddef.h>\nint LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) { (void)data; (void)size; return 0; }" STORIES_HAVE_LIBFUZZER)
	# The C++ core makes CMake use the C++ driver to link fuzzers; checking only the C driver can accept a toolchain that fails at the real link step.
	check_cxx_source_compiles("#include <cstdint>\n#include <cstddef>\nextern \"C\" int LLVMFuzzerTestOneInput(const std::uint8_t *data, std::size_t size) { (void)data; (void)size; return 0; }" STORIES_HAVE_CXX_LIBFUZZER)
	cmake_pop_check_state()
	if(NOT STORIES_HAVE_LIBFUZZER)
		message(FATAL_ERROR "Fuzzing requires Clang with libFuzzer, ASan, and UBSan. Select a full LLVM Clang installation with CMAKE_C_COMPILER.")
	endif()
	if(NOT STORIES_HAVE_CXX_LIBFUZZER)
		message(FATAL_ERROR "Fuzzers link with the C++ driver (${CMAKE_CXX_COMPILER}), which must provide libFuzzer, ASan, and UBSan. Select the matching full LLVM C++ compiler with CMAKE_CXX_COMPILER or CXX.")
	endif()
	add_library(stories-fuzz-core STATIC ${stories_core_sources})
	add_dependencies(stories-fuzz-core stories-entities)
	target_include_directories(stories-fuzz-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}")
	target_link_libraries(stories-fuzz-core PUBLIC MD4C::MD4C stories-inja stories-options)
	target_compile_options(stories-fuzz-core PRIVATE ${stories_fuzz_flags})
	foreach(target url_resolver link_rewriter config_parser)
		add_executable(fuzz_${target} fuzz/fuzz_${target}.c)
		target_link_libraries(fuzz_${target} PRIVATE stories-fuzz-core)
		target_compile_definitions(fuzz_${target} PRIVATE TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
		target_compile_options(fuzz_${target} PRIVATE ${stories_fuzz_flags})
		target_link_options(fuzz_${target} PRIVATE ${stories_fuzz_link_flags})
		add_test(NAME fuzz-${target} COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/smoke.py" "$<TARGET_FILE:fuzz_${target}>" "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus/${target}" "${CMAKE_CURRENT_BINARY_DIR}/fuzz-artifacts/${target}" 30)
		set_tests_properties(fuzz-${target} PROPERTIES LABELS fuzz TIMEOUT 50)
	endforeach()
endif()
