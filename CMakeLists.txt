cmake_minimum_required(VERSION 3.20)

include(cmake/SelectCompiler.cmake)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" STORIES_VERSION LIMIT_COUNT 1)
if(NOT STORIES_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+(-[A-Za-z0-9.-]+)?$")
	message(FATAL_ERROR "VERSION must contain a release version such as 1.2.3 or 1.2.3-rc.1")
endif()
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" STORIES_PROJECT_VERSION "${STORIES_VERSION}")
project(stories VERSION "${STORIES_PROJECT_VERSION}" LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build configuration" FORCE)
endif()

include(CTest)
include(GNUInstallDirs)
include(CheckCSourceCompiles)
include(CMakePushCheckState)
include(FindPackageHandleStandardArgs)

option(STORIES_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(STORIES_SANITIZERS "Instrument application code with ASan and UBSan" OFF)
option(STORIES_FUZZING "Build Clang libFuzzer targets with ASan and UBSan" OFF)
option(STORIES_STATIC_DEPENDENCIES "Link CivetWeb and MD4C from static libraries" OFF)

find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)

# Consult pkgconf when present; direct discovery also works without it.
find_program(STORIES_PKGCONF NAMES pkgconf)
if(STORIES_PKGCONF)
	set(PKG_CONFIG_EXECUTABLE "${STORIES_PKGCONF}")
	find_package(PkgConfig QUIET)
	if(PKG_CONFIG_FOUND)
		pkg_check_modules(CIVETWEB_PC QUIET civetweb)
		pkg_check_modules(MD4C_PC QUIET md4c)
	endif()
endif()

if(STORIES_STATIC_DEPENDENCIES)
	set(stories_library_suffixes "${CMAKE_FIND_LIBRARY_SUFFIXES}")
	set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
endif()
find_path(CIVETWEB_INCLUDE_DIR civetweb.h HINTS ${CIVETWEB_PC_INCLUDE_DIRS})
find_library(CIVETWEB_LIBRARY NAMES civetweb HINTS ${CIVETWEB_PC_LIBRARY_DIRS})
find_path(MD4C_INCLUDE_DIR md4c.h HINTS ${MD4C_PC_INCLUDE_DIRS})
find_library(MD4C_LIBRARY NAMES md4c HINTS ${MD4C_PC_LIBRARY_DIRS})
if(STORIES_STATIC_DEPENDENCIES)
	set(CMAKE_FIND_LIBRARY_SUFFIXES "${stories_library_suffixes}")
endif()
find_package_handle_standard_args(CivetWeb REQUIRED_VARS CIVETWEB_INCLUDE_DIR CIVETWEB_LIBRARY)
find_package_handle_standard_args(MD4C REQUIRED_VARS MD4C_INCLUDE_DIR MD4C_LIBRARY)
if(NOT CivetWeb_FOUND OR NOT MD4C_FOUND)
	message(FATAL_ERROR "Install CivetWeb and MD4C development headers and libraries")
endif()
add_library(CivetWeb::CivetWeb UNKNOWN IMPORTED)
set_target_properties(CivetWeb::CivetWeb PROPERTIES IMPORTED_LOCATION "${CIVETWEB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${CIVETWEB_INCLUDE_DIR}")
add_library(MD4C::MD4C UNKNOWN IMPORTED)
set_target_properties(MD4C::MD4C PROPERTIES IMPORTED_LOCATION "${MD4C_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MD4C_INCLUDE_DIR}")
mark_as_advanced(CIVETWEB_INCLUDE_DIR CIVETWEB_LIBRARY MD4C_INCLUDE_DIR MD4C_LIBRARY STORIES_PKGCONF)

cmake_push_check_state(RESET)
set(CMAKE_REQUIRED_LIBRARIES CivetWeb::CivetWeb Threads::Threads)
check_c_source_compiles("#include <civetweb.h>\nint main(void) { return mg_get_server_ports(0, 0, 0); }" STORIES_HAVE_SERVER_PORTS)
if(NOT STORIES_HAVE_SERVER_PORTS)
	message(FATAL_ERROR "CivetWeb must provide mg_get_server_ports(). Install CivetWeb 1.16 or newer.")
endif()
set(CMAKE_REQUIRED_LIBRARIES MD4C::MD4C)
check_c_source_compiles("#include <md4c.h>\nint main(void) { return MD_FLAG_TASKLISTS == 0; }" STORIES_HAVE_TASKLISTS)
if(NOT STORIES_HAVE_TASKLISTS)
	message(FATAL_ERROR "MD4C must support task lists. Install MD4C 0.4.8 or newer.")
endif()
cmake_pop_check_state()

add_library(stories-options INTERFACE)
target_compile_definitions(stories-options INTERFACE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700 _FILE_OFFSET_BITS=64)
if(APPLE)
	# Darwin hides O_NOFOLLOW behind its native feature namespace.
	target_compile_definitions(stories-options INTERFACE _DARWIN_C_SOURCE)
endif()
target_compile_options(stories-options INTERFACE -Wall -Wextra -Wpedantic)
if(STORIES_WARNINGS_AS_ERRORS)
	target_compile_options(stories-options INTERFACE -Werror)
endif()
if(STORIES_SANITIZERS)
	cmake_push_check_state(RESET)
	set(CMAKE_REQUIRED_FLAGS "-fsanitize=address,undefined")
	set(CMAKE_REQUIRED_LINK_OPTIONS -fsanitize=address,undefined)
	check_c_source_compiles("int main(void) { return 0; }" STORIES_HAVE_SANITIZERS)
	cmake_pop_check_state()
	if(NOT STORIES_HAVE_SANITIZERS)
		message(FATAL_ERROR "The selected compiler must provide ASan and UBSan when STORIES_SANITIZERS is enabled")
	endif()
	target_compile_options(stories-options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
	target_link_options(stories-options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
endif()

configure_file(cmake/build_config.h.in build_config.h @ONLY)
configure_file(VERSION stories-version.txt COPYONLY)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/entities.inc" COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_entities.py" "${CMAKE_CURRENT_BINARY_DIR}/entities.inc" DEPENDS tools/generate_entities.py VERBATIM)
add_custom_target(stories-entities DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/entities.inc")
add_library(stories-mustach STATIC third_party/mustach/mustach.c)
target_include_directories(stories-mustach SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mustach")
target_compile_definitions(stories-mustach PRIVATE _POSIX_C_SOURCE=200809L)
# Keep upstream code separate from project warnings and analyzer policies.
set_target_properties(stories-mustach PROPERTIES C_CLANG_TIDY "")
if(STORIES_SANITIZERS)
	target_compile_options(stories-mustach PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
endif()
set(stories_core_sources src/buffer.c src/config.c src/url.c src/document.c src/entities.c src/page_template.c src/render.c)
add_library(stories-core STATIC ${stories_core_sources})
add_dependencies(stories-core stories-entities)
target_include_directories(stories-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(stories-core PUBLIC MD4C::MD4C stories-mustach stories-options)
add_executable(stories src/main.c src/server.c)
target_link_libraries(stories PRIVATE stories-core CivetWeb::CivetWeb Threads::Threads)

install(TARGETS stories RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES assets/pico.min.css assets/LICENSE.pico.md DESTINATION "${CMAKE_INSTALL_DATADIR}/stories/assets")
install(FILES templates/page.html DESTINATION "${CMAKE_INSTALL_DATADIR}/stories/templates")
install(FILES third_party/mustach/LICENSE.txt DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/stories/mustach")
install(FILES examples/stories.conf.example examples/docs.conf.example examples/stories.freebsd examples/nginx.conf DESTINATION "${CMAKE_INSTALL_DATADIR}/stories/examples")
install(DIRECTORY docs/ DESTINATION "${CMAKE_INSTALL_DATADIR}/stories/docs" FILES_MATCHING PATTERN "*.md")
install(FILES LICENSE.md DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/stories")

if(BUILD_TESTING)
	add_executable(test-subsystems tests/test_subsystems.c)
	target_link_libraries(test-subsystems PRIVATE stories-core)
	target_compile_definitions(test-subsystems PRIVATE TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
	add_test(NAME subsystems COMMAND test-subsystems)
	add_test(NAME packaging-arguments COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_package_release.py")
	add_test(NAME compiler-selection COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compiler_selection.py" "${CMAKE_COMMAND}")
	set_tests_properties(subsystems packaging-arguments compiler-selection PROPERTIES LABELS unit)
	add_test(NAME http COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py" "$<TARGET_FILE:stories>")
	set_tests_properties(http PROPERTIES LABELS integration TIMEOUT 90)
endif()

if(STORIES_FUZZING)
	enable_testing()
	set(stories_fuzz_flags -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined -UNDEBUG -g -O1)
	set(stories_fuzz_link_flags -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=undefined)
	cmake_push_check_state(RESET)
	set(CMAKE_REQUIRED_FLAGS "-fsanitize=fuzzer-no-link,address,undefined")
	set(CMAKE_REQUIRED_LINK_OPTIONS ${stories_fuzz_link_flags})
	check_c_source_compiles("#include <stdint.h>\n#include <stddef.h>\nint LLVMFuzzerTestOneInput(const uint8_t *data, size_t size) { (void)data; (void)size; return 0; }" STORIES_HAVE_LIBFUZZER)
	cmake_pop_check_state()
	if(NOT STORIES_HAVE_LIBFUZZER)
		message(FATAL_ERROR "Fuzzing requires Clang with libFuzzer, ASan, and UBSan. Select a full LLVM Clang installation with CMAKE_C_COMPILER.")
	endif()
	add_library(stories-fuzz-core STATIC ${stories_core_sources})
	add_dependencies(stories-fuzz-core stories-entities)
	target_include_directories(stories-fuzz-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}")
	target_link_libraries(stories-fuzz-core PUBLIC MD4C::MD4C stories-mustach stories-options)
	target_compile_options(stories-fuzz-core PRIVATE ${stories_fuzz_flags})
	foreach(target url_resolver link_rewriter config_parser)
		add_executable(fuzz_${target} fuzz/fuzz_${target}.c)
		target_link_libraries(fuzz_${target} PRIVATE stories-fuzz-core)
		target_compile_definitions(fuzz_${target} PRIVATE TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
		target_compile_options(fuzz_${target} PRIVATE ${stories_fuzz_flags})
		target_link_options(fuzz_${target} PRIVATE ${stories_fuzz_link_flags})
		add_test(NAME fuzz-${target} COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/smoke.py" "$<TARGET_FILE:fuzz_${target}>" "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus/${target}" "${CMAKE_CURRENT_BINARY_DIR}/fuzz-artifacts/${target}" 30)
		set_tests_properties(fuzz-${target} PROPERTIES LABELS fuzz TIMEOUT 50)
	endforeach()
endif()
